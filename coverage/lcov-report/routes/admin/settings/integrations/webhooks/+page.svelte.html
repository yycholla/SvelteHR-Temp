	function getActionBadgeColor(status: string): string {
		switch (status.toLowerCase()) {

	function formatDuration(ms: number | null): string {
		if (!ms) return 'N/A';

		const minutes = Math.floor(seconds / 60);
		const remainingSeconds = seconds % 60;
