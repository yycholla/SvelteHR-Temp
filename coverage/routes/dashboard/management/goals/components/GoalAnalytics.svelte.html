	function getDepartmentColor(deptName: string): string {
		const colors = [

			'bg-green-500',
			'bg-purple-500',
			'bg-orange-500',
			'bg-indigo-500',
			'bg-pink-500'
		];
		const index = deptName.length % colors.length;
		return colors[index];
	}
